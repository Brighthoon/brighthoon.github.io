from datetime import datetime

class ClimbingProblem(Base):
    """
    원본 영상 및 암장 메타데이터 관리 테이블
    """
    __tablename__ = 'climbing_problems'

    id = Column(String, primary_key=True)      # PRB-0001 (UUID 또는 Hash)
    
    # 1. 메타데이터 (사람이 입력하거나 외부 API 연동)
    gym_name = Column(String)                  # 암장 이름
    difficulty = Column(String)                # 난이도 (V5, 파랑 등)
    
    # 2. 파이프라인 및 원본 파일 정보
    status = Column(String, default="PENDING") # PENDING, PROCESSING, COMPLETED, FAILED
    base_folder_path = Column(String)          # 데이터가 저장될 루트 폴더 경로
    video_filename = Column(String)            # 원본 영상 파일명
    
    # 3. 시간 정보
    created_at = Column(DateTime, default=datetime.utcnow)
    updated_at = Column(DateTime, default=datetime.utcnow, onupdate=datetime.utcnow)

    # 1:N 관계 (하나의 영상에 여러 모델 추론 결과 존재 가능)
    hold_results = relationship("HoldSegResult", back_populates="problem")
    pose_results = relationship("PoseKptResult", back_populates="problem")

class HoldSegResult(Base):
    """
    홀드 세그멘테이션 추론 결과 관리 테이블
    """
    __tablename__ = 'hold_seg_results'

    id = Column(String, primary_key=True)      # HLD-0001
